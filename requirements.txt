# Python packages the detector needs that do NOT come from apt.
#
# This is a list of direct dependencies, not a `pip freeze`. The previous file
# was freeze output: 35 pinned packages including matplotlib, polars, scipy and
# Flask's entire transitive tree, almost none of which anything in this repo
# imports. A freeze also pins the children, so upgrading one real dependency
# turns into a resolver conflict against a dozen pins nobody chose.
#
# What the detector imports, and where each piece comes from:
#
#   rclpy, std_msgs, two_towers.msg   apt, via ros-jazzy-*. NOT in this file
#                                     and not installable from PyPI.
#   cv2                               opencv-python, below
#   flask                             below
#   ultralytics (YOLO)                below; resolves torch and torchvision
#                                     itself, so neither is pinned here
#
# ---------------------------------------------------------------------------
# Creating the venv
# ---------------------------------------------------------------------------
#
#   python3 -m venv --system-site-packages venv
#
# The flag is mandatory, not a convenience. ROS 2 installs its Python packages
# into /opt/ros/jazzy/lib/python3.12/site-packages and there is no pip package
# for rclpy. A plain `python3 -m venv venv` produces an environment that cannot
# see any of it, and the detector dies on `import rclpy` at line 42 before it
# ever reaches YOLO. scripts/bootstrap_pi.sh creates the venv correctly.

# ---------------------------------------------------------------------------
# numpy is capped below 2.0 deliberately, and this pin is load-bearing.
#
# Ubuntu 24.04 ships numpy 1.26, and every ROS 2 Jazzy Python C extension on
# the system is compiled against that ABI -- including the rosidl-generated
# two_towers.msg modules that colcon builds on the Pi itself. Because the venv
# is created with --system-site-packages, a pip-installed numpy 2.x shadows the
# apt one for those extensions and they fail at import with
#
#   AttributeError: _ARRAY_API not found
#   A module that was compiled using NumPy 1.x cannot be run in NumPy 2.x
#
# which surfaces as the detector failing to import its own message types. Raise
# this cap only together with the system numpy.
# ---------------------------------------------------------------------------
numpy>=1.26,<2

ultralytics>=8.3,<9
flask>=3.0,<4

# Deliberately opencv-python and NOT opencv-python-headless, even though the
# towers are headless servers and nothing here opens a window.
#
# ultralytics declares a hard dependency on opencv-python. Installing the
# headless wheel satisfies `import cv2`, but leaves that dependency unmet, so
# the next `pip install -r requirements.txt` reinstalls the GUI wheel on top
# and quietly undoes it. The cost of keeping opencv-python is two system
# libraries (libgl1, libglib2.0-0t64) that scripts/bootstrap_pi.sh installs;
# without them `import cv2` fails with "ImportError: libGL.so.1".
#
# The version range interacts with the numpy cap above: opencv-python 4.12+
# requires numpy>=2, so the resolver backtracks to 4.11.0.86, which is the
# newest build that still supports numpy 1.x. That is intended, not a
# coincidence -- raising the numpy cap would let opencv pull numpy 2 in
# through the side door and break the generated message modules.
opencv-python>=4.9,<5

# colcon-core requires setuptools<80. Nothing here asks for setuptools
# directly, but pip pulls a much newer one in transitively, and because the
# venv is --system-site-packages it then shadows the apt setuptools that
# colcon runs against. The symptom is `colcon build` breaking only after you
# activate the venv, which is exactly what the README tells you to do.
setuptools<80
